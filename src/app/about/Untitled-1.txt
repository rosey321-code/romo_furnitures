    <!-- JavaScript for Dynamic Mouse Tracking -->
    <script>
        const card = document.getElementById('parallaxCard');
        const bg = document.getElementById('cardBg');

        card.addEventListener('mousemove', (e) => {
            const cardRect = card.getBoundingClientRect();
            
            // Mouse ki position card ke center se calculate karna
            const cardWidth = cardRect.width;
            const cardHeight = cardRect.height;
            const mouseX = e.clientX - cardRect.left - cardWidth / 2;
            const mouseY = e.clientY - cardRect.top - cardHeight / 2;
            
            // Max rotation angles (jitna rotate karwana ho)
            const angleX = -(mouseY / cardHeight) * 20; // Up/Down tilt
            const angleY = (mouseX / cardWidth) * 20;  // Left/Right tilt
            
            // Background Image ka inverse shift (Parallax depth ke liye)
            const shiftX = (mouseX / cardWidth) * 15;
            const shiftY = (mouseY / cardHeight) * 15;

            // 3D Matrix apply karna smooth movements ke sath
            card.style.transform = `rotateX(${angleX}deg) rotateY(${angleY}deg)`;
            bg.style.transform = `scale(1.1) translateX(${-shiftX}px) translateY(${-shiftY}px)`;
        });

        // Mouse leave hone par card ka wapis apni jagah smooth aa jana
        card.addEventListener('mouseleave', () => {
            card.style.transform = 'rotateX(0deg) rotateY(0deg)';
            bg.style.transform = 'scale(1) translateX(0px) translateY(0px)';
        });
    </script>
